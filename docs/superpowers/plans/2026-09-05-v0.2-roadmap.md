`_save_cp` 方法改造为带观测 (返回前调回调):

```python
def _save_cp(self, task_id: str, dag: DAG, state: dict, done_stages: list,
             statuses: dict, producers: dict | None = None,
             stage_name: str | None = None) -> None:
    if self.checkpoint_store is None:
        return
    try:
        from .checkpoint import workflow_hash

        cp = Checkpoint(
            task_id=task_id,
            dag_name=dag.name,
            workflow_hash=workflow_hash(dag),
            stage_statuses=dict(statuses),
            state=state,
            done_stages=list(done_stages),
            producers=dict(producers or {}),
        )
        # M1: size + save 计时
        import time as _t, json as _j
        _body = _j.dumps(cp.to_dict(), ensure_ascii=False)
        size_bytes = len(_body.encode("utf-8"))
        _t0 = _t.perf_counter()
        self.checkpoint_store.save(cp)
        save_ms = (_t.perf_counter() - _t0) * 1000
        if self.on_checkpoint is not None:
            try:
                self.on_checkpoint(CheckpointStats(
                    size_bytes=size_bytes,
                    save_ms=save_ms,
                    stage_name=stage_name,
                    task_id=task_id,
                ))
            except Exception:
                logger.exception("on_checkpoint callback raised (non-fatal)")
    except Exception:
        logger.exception("task=%s checkpoint 保存失败 (non-fatal)", task_id)
```

所有 `_save_cp` 调用点加 `stage_name=name` (run 内成功路径) 或 `stage_name=None` (run 终态清 checkpoint):

- `done_stages.append(name)` 之前那次调用 → `stage_name=name`
- run 失败 `_save_cp(... done_stages, stage_statuses, producers)` → `stage_name=None`
- run 成功结束 `self.checkpoint_store.delete(...)` 之前改为 `_save_cp(... stage_name=None)` (统一走 save, 让回调收 size; 然后 Runtime 再 delete)

实际方案: 不要在 run 终态额外调 _save_cp; 改为 run 终态 delete 前手算 size (或: 简化为 callback 只在每个 node done 后触发, run 终态清 checkpoint 不触发 — 这就是现状, 不必改)。**最终方案** — callback 只在 `_save_cp` 真正 save 时触发, run done 的 delete 不触发回调:

```python
# 在 runtime.run() done 路径 — 不改
if self.checkpoint_store is not None:
    self.checkpoint_store.delete(task_id)
return result
```

```python
def replay_from(self, checkpoint: dict, *, run_from: str | None = None) -> "TestPipe":
    """从 checkpoint dict 恢复: 已完成 stage 用原 delta mock, 余下 stage 真跑.

    Args:
        checkpoint: Checkpoint.to_dict() 格式 (含 stage_deltas 字段)
        run_from: 真跑起点 (默认 = 全 mock 跑通 = 纯回归)
    """
    cp = Checkpoint.from_dict(checkpoint) if "task_id" in checkpoint else checkpoint
    # 兼容: 用户传 Checkpoint.to_dict() (dict) 或 Checkpoint 实例
    if isinstance(cp, Checkpoint):
        stage_deltas = cp.stage_deltas
        done_stages = list(cp.done_stages)
        state = dict(cp.state)
    else:
        stage_deltas = cp.get("stage_deltas") or {}
        if not stage_deltas:
            raise RuntimeError(
                "replay_from 需要含 stage_deltas 的 checkpoint; "
                "旧 cp (v0.1.x) 不带此字段, 请重跑一次生成新 cp"
            )
        done_stages = list(cp.get("done_stages") or [])
        state = dict(cp.get("state") or {})

    # 注册每 done stage 为 mock (返回其原 delta)
    for stage_name, delta in in stage_deltas.items():
        self.mock(stage_name, lambda _state, _d=delta: dict(_d))

    # 设 run_from: 跳过其上游所有 stage 的真跑 (改成 mock 已 done; run 顺序跳过)
    if run_from is not None:
        if run_from not in self.dag.stages:
            raise ValueError(f"run_from '{run_from}' 不在 DAG '{self.dag.name}' 的 stage 中")
        # TestPipe.run 顺序走 topo_order; 已 done stage 已被 mock 覆盖 (返回 delta)
        # run_from = 让 TestPipe 把该 stage 之前的 stage 视作 "done" 不执行?
        # 简化: TestPipe 默认全跑, mock 已 done stage 后其输出与 cp.state 一致
        # 后续 stage 读到正确 state, 真跑结果与原一致
        # 不需要额外标志 — mock + run 自然 work
        pass

    return self
```

```python
async def run_stage(
    self,
    dag: DAG,
    task_id: str,
    stage_name: str,
    *,
    initial_state: dict | None = None,
    resume: bool = True,
    deadline: float | None = None,
) -> RunResult:
    """从 checkpoint 恢复 state, 只执行 stage_name (依赖须在 done_stages 内).

    不修改 API 冻结面: 与 run() 复用同一 state merge / checkpoint / retry 逻辑.
    失败行为: 缺 checkpoint / hash mismatch / stage 依赖未完成 → 明确 raise.
    """
    dag.validate()
    if stage_name not in dag.stages:
        raise ValueError(f"stage '{stage_name}' 不在 DAG '{dag.name}' 中")

    if self.checkpoint_store is None:
        raise RuntimeError("run_stage 需要 CheckpointStore; 传 Runtime(checkpoint_store=...)")

    # 加载 cp (hash mismatch raise)
    cp = self.checkpoint_store.load_compatible(task_id, dag)
    if cp is None:
        raise RuntimeError(f"task '{task_id}' 无 checkpoint; 先跑一次 run() 生成 cp")

    state: dict = dict(cp.state)
    done_stages: list[str] = list(cp.done_stages)
    producers: dict[str, str] = dict(cp.producers)

    # 依赖校验
    target_stage = dag.stages[stage_name]
    missing = [d for d in target_stage.depends_on if d not in done_stages]
    if missing:
        raise RuntimeError(
            f"stage '{stage_name}' 依赖 {missing} 未在 done_stages, "
            f"不能单跑。先跑这些依赖或换 stage_name"
        )

    run_deadline = deadline or (time.time() + self.default_timeout if self.default_timeout else None)
    stage_deadline = None
    if target_stage.timeout is not None:
        stage_deadline = time.time() + target_stage.timeout
    if run_deadline is not None:
        stage_deadline = (
            min(stage_deadline, run_deadline) if stage_deadline is not None else run_deadline
        )

    self._stage_deltas = dict(cp.stage_deltas)
    status, state, err, producers = await self._run_stage(
        dag, target_stage.fn, stage_name, task_id, state,
        target_stage.retries, stage_deadline, producers,
        legacy_resume=not bool(cp.producers),
    )

    result = RunResult(task_id, dag.name)
    result.status = status
    result.state = state
    result.stage_statuses = {stage_name: status}
    result.error = err

    # 保存更新后 cp (覆盖原 cp, 含新 stage_done + 新 delta)
    if status == "done":
        new_done = done_stages + [stage_name]
        # stage_deltas 增量更新
        if not hasattr(self, "_stage_deltas") or not self._stage_deltas:
            self._stage_deltas = dict(cp.stage_deltas)
        self._save_cp(task_id, dag, state, new_done,
                     {stage_name: status}, producers,
                     stage_name=stage_name)
        # _save_cp 用 self._stage_deltas 持久化新 delta
    return result
```

```python
@cli.command()
@click.option("--task-id", required=True)
@click.option("--stage", "stage_name", required=True, help="目标 stage 名")
@click.option("--patch", "patch_path", default=None,
              help="可选 patch 文件路径 (import 后 monkey-patch dag/caller)")
@click.argument("dag_file", type=click.Path(exists=True))
def replay(dag_file: str, task_id: str, stage_name: str, patch_path: str | None):
    """从 checkpoint 恢复, 单 stage 重放.

    DAG_FILE: 包含 `dag = DAG(...)` 变量的 .py 文件路径.
    """
    import asyncio
    import importlib.util

    spec = importlib.util.spec_from_file_location("cli_dag_module", dag_file)
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    dag = mod.dag

    # 加载 patch (import 副作用)
    if patch_path:
        spec_p = importlib.util.spec_from_file_location("cli_patch", patch_path)
        mod_p = importlib.util.module_from_spec(spec_p)
        spec_p.loader.exec_module(mod_p)
        # 可选: 若 patch 暴露 patch(dag) 函数, 调一下
        if hasattr(mod_p, "patch"):
            mod_p.patch(dag)

    storage_root = _storage_root()
    rt = Runtime(checkpoint_store=CheckpointStore(FileStorage(storage_root)))
    result = asyncio.run(rt.run_stage(dag, task_id, stage_name))
    click.echo(json.dumps({
        "task_id": result.task_id,
        "status": result.status,
        "stage_statuses": result.stage_statuses,
        "state": result.state,
        "error": result.error,
    }, ensure_ascii=False, indent=2))
    if result.status != "done":
        sys.exit(1)
```
